"""Presupuestos (7.6): simulación con ajustes por rubro desde un mes, y planilla para cargar o exportar.

Simulación (decisión del director, 18/11/2026): cada ajuste es un porcentaje (positivo o negativo) sobre una
cuenta o un rubro, desde un mes del presupuesto en adelante. Si a una línea le corresponden varios ajustes, se
acumulan (julio +4 % y octubre +3 %: desde octubre, base × 1,04 × 1,03). El resultado se guarda como un
escenario nuevo; el presupuesto de origen no cambia. Sin ajustes, todo queda igual.

Planilla: una hoja con Código, Cuenta, Local, Centro y los 12 meses. Local y Centro vacíos = la empresa en
total. Las filas con todos los importes en cero se ignoran. Se importa todo o nada.
"""
import datetime as _dt

from .base import SECCIONES_ER, SECCIONES_GASTO, ErrorContable, fmt_importe, parse_importe

MESES_CORTOS = ["ene", "feb", "mar", "abr", "may", "jun", "jul", "ago", "set", "oct", "nov", "dic"]
COLUMNAS = ["Código", "Cuenta", "Local", "Centro"]


def _mes_n(inicio, n):
    d = _dt.date.fromisoformat(inicio)
    m = d.month - 1 + n
    return d.replace(year=d.year + m // 12, month=m % 12 + 1, day=1)


def etiquetas_meses(inicio):
    return [f"{MESES_CORTOS[_mes_n(inicio, k).month - 1]}-{_mes_n(inicio, k).year}" for k in range(12)]


def _texto_codigo(v):
    if v is None:
        return ""
    if isinstance(v, float) and v.is_integer():
        v = int(v)
    return str(v).strip()


def _importe(v):
    if v is None or v == "":
        return 0
    if isinstance(v, (int, float)) and not isinstance(v, bool):
        return int(round(v * 100))
    return parse_importe(str(v))


class PresupuestosPlanillaMixin:
    # ------------------------------------------------------------------ simulación
    def simular_presupuesto(self, pid, nombre, ajustes, descripcion=""):
        """ajustes: lista de dict(cuenta_id, mes (1 a 12, mes del presupuesto), pct). Devuelve el id nuevo."""
        p = self.presupuesto(pid)
        emp = p["empresa_id"]
        mapa = self.mapa_cuentas(emp)
        reglas = []
        for k, a in enumerate(ajustes, 1):
            c = mapa.get(a["cuenta_id"])
            if not c or c["seccion_ef"] not in SECCIONES_ER:
                raise ErrorContable(f"Ajuste {k}: elija una cuenta o un rubro de ingresos o gastos.")
            mes = int(a["mes"])
            if not 1 <= mes <= 12:
                raise ErrorContable(f"Ajuste {k}: el mes debe ser uno de los 12 del presupuesto.")
            pct = float(a["pct"])
            if pct <= -100:
                raise ErrorContable(f"Ajuste {k}: un ajuste de {pct:g} % dejaría los importes en cero o negativos.")
            reglas.append((c["codigo"], mes, pct))
        etiquetas = etiquetas_meses(p["inicio"])
        resumen = "; ".join(f"{cod} {pct:+g} % desde {etiquetas[mes - 1]}" for cod, mes, pct in reglas) or "sin ajustes"
        nuevo = self.crear_presupuesto(emp, nombre, None, descripcion or f"Simulación sobre «{p['nombre']}»: {resumen}",
                                       f"simulación de «{p['nombre']}»")
        filas = []
        for f in self.con.execute("SELECT i.*, c.codigo FROM presupuesto_items i JOIN cuentas c ON c.id=i.cuenta_id "
                                  "WHERE presupuesto_id=?", (pid,)).fetchall():
            factor = 1.0
            for cod, mes, pct in reglas:
                if f["codigo"].startswith(cod) and f["mes"] >= mes:
                    factor *= 1 + pct / 100
            filas.append((nuevo, f["cuenta_id"], f["cc_id"], f["local_id"], f["mes"], int(round(f["importe"] * factor))))
        with self.con:
            self.con.execute("UPDATE presupuestos SET inicio=? WHERE id=?", (p["inicio"], nuevo))
            self.con.executemany("INSERT INTO presupuesto_items(presupuesto_id,cuenta_id,cc_id,local_id,mes,importe) "
                                 "VALUES (?,?,?,?,?,?)", filas)
        return nuevo

    def totales_mensuales(self, pid):
        """(ingresos[12], gastos[12]) en sentido natural, para comparar escenarios."""
        p = self.presupuesto(pid)
        mapa = self.mapa_cuentas(p["empresa_id"])
        ing, gas = [0] * 12, [0] * 12
        for f in self.con.execute("SELECT cuenta_id, mes, importe FROM presupuesto_items WHERE presupuesto_id=?",
                                  (pid,)):
            sec = mapa[f["cuenta_id"]]["seccion_ef"]
            (gas if sec in SECCIONES_GASTO else ing)[f["mes"] - 1] += f["importe"]
        return ing, gas

    # ------------------------------------------------------------------ planilla
    def planilla_presupuesto(self, emp, ruta, pid=None):
        """Planilla modelo con el plan de cuentas actual (pid=None) o el presupuesto pid exportado."""
        try:
            import openpyxl
            from openpyxl.styles import Alignment, Font, PatternFill
        except ImportError:
            raise ErrorContable("Para generar planillas de Excel instale openpyxl (ver LEEME).")
        e = self.empresa(emp)
        inicio = self.presupuesto(pid)["inicio"] if pid else self._inicio_por_defecto(emp)
        meses = etiquetas_meses(inicio)
        mapa = self.mapa_cuentas(emp)
        cuentas = [c for c in sorted(mapa.values(), key=lambda x: x["codigo"])
                   if c["seccion_ef"] in SECCIONES_ER and c["nivel"] >= 2]
        locales = {l["id"]: l["codigo"] for l in self.locales(emp)}
        centros = {c["id"]: c["codigo"] for c in self.centros_costo(emp)}
        cargadas = {}
        if pid:
            for l in self.lineas_presupuesto(pid):
                cargadas.setdefault(l["cuenta_id"], []).append(l)
        libro = openpyxl.Workbook()
        h = libro.active
        h.title = "Presupuesto"
        h["A1"] = f"Presupuesto · {e['codigo']} {e['nombre']}" + (f" · {self.presupuesto(pid)['nombre']}" if pid else "")
        h["A1"].font = Font(bold=True, size=13)
        h["A2"] = (f"Primer mes: {meses[0]}. Importes en {e['moneda_nacional']}, en positivo (ingreso o gasto "
                   "esperado). Local y Centro vacíos = empresa total; para presupuestar por local o centro copie la "
                   "fila y escriba el código. No cargue un rubro y una de sus cuentas para el mismo local y centro. "
                   "Las filas en cero se ignoran.")
        h["A2"].alignment = Alignment(wrap_text=True, vertical="top")
        h.merge_cells("A2:P2")
        h.row_dimensions[2].height = 45
        fila = 4
        cab = COLUMNAS + meses + ["Total"]
        for j, t in enumerate(cab, 1):
            c = h.cell(fila, j, t)
            c.font = Font(bold=True, color="FFFFFF")
            c.fill = PatternFill("solid", fgColor="1F3864")
            c.alignment = Alignment(horizontal="center")
        gris = PatternFill("solid", fgColor="E7E6E6")
        for c in cuentas:
            lineas = cargadas.get(c["id"]) or [None]
            for l in lineas:
                fila += 1
                h.cell(fila, 1, c["codigo"])
                h.cell(fila, 2, ("   " * (c["nivel"] - 2)) + c["nombre"])
                if l:
                    h.cell(fila, 3, locales.get(l["local_id"], ""))
                    h.cell(fila, 4, centros.get(l["cc_id"], ""))
                    for k, v in enumerate(l["importes"]):
                        if v:
                            h.cell(fila, 5 + k, abs(v) / 100).number_format = "#,##0.00"
                for k in range(12):
                    h.cell(fila, 5 + k).number_format = "#,##0.00"
                h.cell(fila, 17, f"=SUM(E{fila}:P{fila})").number_format = "#,##0.00"
                if not c["imputable"]:
                    for j in range(1, 18):
                        h.cell(fila, j).fill = gris
                    h.cell(fila, 2).font = Font(bold=True)
        h.column_dimensions["A"].width = 12
        h.column_dimensions["B"].width = 42
        for col in "CD":
            h.column_dimensions[col].width = 8
        for j in range(5, 18):
            h.column_dimensions[openpyxl.utils.get_column_letter(j)].width = 12
        h.freeze_panes = "E5"
        d = libro.create_sheet("Locales y centros")
        d.append(["Tipo", "Código", "Nombre"])
        for l in self.locales(emp):
            d.append(["Local", l["codigo"], l["nombre"]])
        for c in self.centros_costo(emp):
            d.append(["Centro de costos", c["codigo"], c["nombre"]])
        d.column_dimensions["C"].width = 30
        try:
            libro.save(ruta)
        except OSError as x:
            raise ErrorContable(f"No se pudo guardar la planilla en {ruta} ({x}).")
        return ruta

    def leer_planilla_presupuesto(self, emp, ruta):
        """Devuelve (lineas, ignoradas, errores). lineas: dict(cuenta_id, local_id, cc_id, importes[12], fila)."""
        try:
            import openpyxl
        except ImportError:
            raise ErrorContable("Para leer planillas de Excel instale openpyxl (ver LEEME).")
        try:
            libro = openpyxl.load_workbook(ruta, read_only=True, data_only=True)
        except Exception as x:
            raise ErrorContable(f"No se pudo leer la planilla ({x}).")
        h = libro["Presupuesto"] if "Presupuesto" in libro.sheetnames else libro.worksheets[0]
        filas = list(h.iter_rows(values_only=True))
        cab = next((k for k, f in enumerate(filas) if f and _texto_codigo(f[0]).lower() in ("código", "codigo")), None)
        if cab is None:
            raise ErrorContable("La planilla no tiene la fila de títulos (Código, Cuenta, Local, Centro y los 12 meses).")
        por_codigo = {c["codigo"]: c for c in self.mapa_cuentas(emp).values()}
        locales = {l["codigo"].upper(): l["id"] for l in self.locales(emp)}
        centros = {c["codigo"].upper(): c["id"] for c in self.centros_costo(emp)}
        lineas, ignoradas, errores, vistas = [], 0, [], {}
        for n, f in enumerate(filas[cab + 1:], cab + 2):
            f = list(f) + [None] * 16
            cod = _texto_codigo(f[0])
            if not cod:
                continue
            try:
                importes = [_importe(v) for v in f[4:16]]
            except ErrorContable as x:
                errores.append(f"Fila {n}: {x}")
                continue
            if not any(importes):
                ignoradas += 1
                continue
            c = por_codigo.get(cod)
            if not c:
                errores.append(f"Fila {n}: la cuenta {cod} no existe en el plan de la empresa.")
                continue
            if c["seccion_ef"] not in SECCIONES_ER or c["nivel"] < 2:
                errores.append(f"Fila {n}: {cod} {c['nombre']} no es una cuenta o rubro de ingresos o gastos "
                               "(grupos 4 y 5, desde el segundo nivel).")
                continue
            if any(v < 0 for v in importes):
                errores.append(f"Fila {n}: los importes van en positivo (ingreso o gasto esperado).")
                continue
            loc_t, cc_t = _texto_codigo(f[2]).upper(), _texto_codigo(f[3]).upper()
            if loc_t and loc_t not in locales:
                errores.append(f"Fila {n}: el local {loc_t} no existe.")
                continue
            if cc_t and cc_t not in centros:
                errores.append(f"Fila {n}: el centro de costos {cc_t} no existe.")
                continue
            loc, cc = locales.get(loc_t), centros.get(cc_t)
            clave = (c["id"], loc, cc)
            if clave in vistas:
                errores.append(f"Fila {n}: {cod} con el mismo local y centro ya está en la fila {vistas[clave]}.")
                continue
            for (oid, oloc, occ), ofila in vistas.items():
                otra = self.cuenta(oid)
                if (oloc, occ) == (loc, cc) and (otra["codigo"].startswith(cod) or cod.startswith(otra["codigo"])):
                    errores.append(f"Fila {n}: {cod} y {otra['codigo']} (fila {ofila}) son rubro y cuenta del mismo "
                                   "grupo con el mismo local y centro: cargue uno solo.")
                    break
            else:
                vistas[clave] = n
                lineas.append(dict(cuenta_id=c["id"], local_id=loc, cc_id=cc, importes=importes, fila=n))
        return lineas, ignoradas, errores

    def importar_presupuesto(self, emp, ruta, pid=None, nombre=None, inicio=None):
        """Importa la planilla a un presupuesto nuevo (nombre, inicio) o reemplaza las líneas de pid. Todo o nada."""
        lineas, ignoradas, errores = self.leer_planilla_presupuesto(emp, ruta)
        if errores:
            raise ErrorContable("No se importó nada. Corrija la planilla:\n" + "\n".join(errores[:25])
                                + (f"\n… y {len(errores) - 25} más." if len(errores) > 25 else ""))
        if not lineas:
            raise ErrorContable("La planilla no tiene importes para importar (todas las filas están en cero).")
        nuevo = pid is None
        if nuevo:
            pid = self.crear_presupuesto(emp, nombre, inicio, f"Importado de {ruta.replace(chr(92), '/').rsplit('/', 1)[-1]}",
                                         "planilla")
        previas = self.con.execute("SELECT cuenta_id, cc_id, local_id, mes, importe FROM presupuesto_items "
                                   "WHERE presupuesto_id=?", (pid,)).fetchall()
        try:
            with self.con:
                self.con.execute("DELETE FROM presupuesto_items WHERE presupuesto_id=?", (pid,))
            for l in lineas:
                self.guardar_linea(pid, l["cuenta_id"], l["importes"], l["cc_id"], l["local_id"])
        except ErrorContable:
            if nuevo:
                self.eliminar_presupuesto(pid)
            else:                                  # se restituyen las líneas que tenía
                with self.con:
                    self.con.execute("DELETE FROM presupuesto_items WHERE presupuesto_id=?", (pid,))
                    self.con.executemany("INSERT INTO presupuesto_items(presupuesto_id,cuenta_id,cc_id,local_id,mes,"
                                         "importe) VALUES (?,?,?,?,?,?)", [(pid,) + tuple(r) for r in previas])
            raise
        total = sum(sum(abs(v) for v in l["importes"]) for l in lineas)
        return dict(presupuesto_id=pid, lineas=len(lineas), ignoradas=ignoradas, total=total,
                    texto=f"{len(lineas)} línea(s) importada(s) por $ {fmt_importe(total)}; "
                          f"{ignoradas} fila(s) en cero ignorada(s).")
